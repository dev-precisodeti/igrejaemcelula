cat << 'EOF' > database/schema.sql
-- Modelagem PostgreSQL Multi-Tenant: Igreja em Célula (Estilo Instagram)
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE TYPE user_role AS ENUM (
    'VISITOR', 'MEMBER', 'HOST', 'LEADER_IN_TRAINING',
    'CELL_LEADER', 'SUPERVISOR', 'NETWORK_LEADER', 'CHURCH_ADMIN'
);
CREATE TYPE membership_status AS ENUM ('PENDING', 'ACTIVE', 'BLOCKED');
CREATE TYPE visibility_scope AS ENUM ('CHURCH', 'NETWORK', 'CELL');
CREATE TYPE post_type AS ENUM ('PHOTO', 'CAROUSEL', 'REEL', 'OFFICIAL_NOTICE', 'CELL_REPORT');

CREATE TABLE churches (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(150) NOT NULL,
    slug VARCHAR(60) UNIQUE NOT NULL,
    invite_code VARCHAR(12) UNIQUE NOT NULL,
    logo_url TEXT,
    city VARCHAR(100),
    state VARCHAR(50),
    require_member_approval BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    firebase_uid VARCHAR(128) UNIQUE,
    username VARCHAR(40) UNIQUE NOT NULL,
    full_name VARCHAR(120) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    avatar_url TEXT,
    bio VARCHAR(250),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE networks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    church_id UUID NOT NULL REFERENCES churches(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    color_hex VARCHAR(7) NOT NULL DEFAULT '#2563EB',
    leader_id UUID REFERENCES users(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE cells (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    church_id UUID NOT NULL REFERENCES churches(id) ON DELETE CASCADE,
    network_id UUID NOT NULL REFERENCES networks(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    leader_id UUID REFERENCES users(id) ON DELETE SET NULL,
    meeting_day SMALLINT CHECK (meeting_day BETWEEN 0 AND 6),
    meeting_time TIME,
    neighborhood VARCHAR(100),
    latitude DECIMAL(9,6),
    longitude DECIMAL(9,6),
    accepting_new_members BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE church_memberships (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    church_id UUID NOT NULL REFERENCES churches(id) ON DELETE CASCADE,
    network_id UUID REFERENCES networks(id) ON DELETE SET NULL,
    cell_id UUID REFERENCES cells(id) ON DELETE SET NULL,
    role user_role NOT NULL DEFAULT 'VISITOR',
    status membership_status NOT NULL DEFAULT 'ACTIVE',
    joined_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE (user_id, church_id)
);

CREATE TABLE posts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    church_id UUID NOT NULL REFERENCES churches(id) ON DELETE CASCADE,
    author_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    network_id UUID REFERENCES networks(id) ON DELETE SET NULL,
    cell_id UUID REFERENCES cells(id) ON DELETE SET NULL,
    visibility visibility_scope NOT NULL DEFAULT 'CHURCH',
    post_type post_type NOT NULL DEFAULT 'PHOTO',
    caption TEXT,
    likes_count INT NOT NULL DEFAULT 0,
    comments_count INT NOT NULL DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);
EOF

# Igreja em Célula (App Social Estilo Instagram Multi-Tenant)

Aplicativo em **Flutter** (Mobile & Web) com arquitetura **Multi-Tenant** para **Igrejas em Células**, integrado ao **Google Cloud Run** e **PostgreSQL**.

## Funcionalidades Principais
- **Vínculo Obrigatório por Igreja (Isolamento Multi-Tenant)**: Cada membro vê apenas o conteúdo, redes e células da sua própria igreja.
- **Hierarquia Celular**: Igreja -> Redes (com identidade de cores) -> Supervisões -> Células.
- **Feed e Stories Estilo Instagram**: Filtros por *Toda a Igreja*, *Minha Rede* e *Minha Célula*.
- **Relatório Semanal que vira Post**: O líder registra a chamada e visitantes e publica a foto oficial da célula direto no Feed.
- **Explorar Redes e Células**: Busca de células por Rede, dia da semana e distância.
- **Painel Pastoral (Admin)**: Indicadores em tempo real para Pastores Sênior e Pastores de Rede.
- **Acessibilidade (WCAG AA)**: Contraste dinâmico nas cores das redes, semântica para leitores de tela e alvos de toque mínimos de `48x48dp`.
